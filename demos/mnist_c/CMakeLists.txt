cmake_minimum_required(VERSION 3.13)
option(MNIST_PICO "Build the native RP2350 USB-console demo" OFF)
if(MNIST_PICO)
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT PICO_SDK_PATH)
        message(FATAL_ERROR "Set PICO_SDK_PATH to an installed Raspberry Pi Pico SDK")
    endif()
    if(NOT PICO_BOARD)
        set(PICO_BOARD pico2 CACHE STRING "Board configuration")
    endif()
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
endif()
project(mnist_demo C CXX ASM)
set(CMAKE_C_STANDARD 11)
if(MNIST_PICO)
    pico_sdk_init()
endif()
add_executable(mnist_demo demo.c neural_net.c)
target_include_directories(mnist_demo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/build/generated")
target_compile_options(mnist_demo PRIVATE -O3 -Wall -Wextra -Werror)
if(MNIST_PICO)
    target_compile_definitions(mnist_demo PRIVATE NN_PICO=1 PICO_STACK_SIZE=8192)
    target_link_libraries(mnist_demo pico_stdlib hardware_clocks)
    pico_enable_stdio_usb(mnist_demo 1)
    pico_enable_stdio_uart(mnist_demo 0)
    pico_add_extra_outputs(mnist_demo)
endif()
